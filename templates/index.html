Solidity Smart Contract

`TransactionAuditor.sol` stores immutable audit hashes on Ethereum Sepolia testnet without exposing confidential transaction data on-chain.

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract TransactionAuditor {
    struct AuditRecord {
        string transactionRef;      // Unique business reference (e.g. TXN-1001)
        string auditHash;           // SHA-256 hash of the canonical transaction audit record
        uint256 timestamp;          // On-chain registration timestamp
        address recordedBy;         // Address of auditor system
        bool exists;
    }

    // Mapping from transaction reference to its immutable audit record
    mapping(string => AuditRecord) private records;

    // Events
    event AuditRecordLogged(string indexed transactionRef, string auditHash, uint256 timestamp);

    function logAuditRecord(string memory _txRef, string memory _auditHash) public {
        require(!records[_txRef].exists, "Audit record already exists and is immutable");
        records[_txRef] = AuditRecord(_txRef, _auditHash, block.timestamp, msg.sender, true);
        emit AuditRecordLogged(_txRef, _auditHash, block.timestamp);
    }

    function verifyRecordHash(string memory _txRef, string memory _calcHash) 
        external view returns (bool matches, string memory storedHash, uint256 timestamp) 
    {
        require(records[_txRef].exists, "Transaction reference not found on blockchain");
        AuditRecord memory record = records[_txRef];
        bool isMatching = keccak256(bytes(record.auditHash)) == keccak256(bytes(_calcHash));
        return (isMatching, record.auditHash, record.timestamp);
    }
}

End-to-End System Pipeline

How AI risk analysis and Blockchain tamper auditing complement each other without leaking financial privacy.

1
Transaction Ingestion & Validation

Financial attributes (amount, mechanism, balance ratios, time, anomaly flags) are validated by the Flask web application.

2
Random Forest AI Scoring

Trained Machine Learning model estimates continuous fraud risk probability (0.0 - 1.0) and assigns category (Low, Medium, High Risk).

3
Canonical SHA-256 Hashing

Deterministic string serialization generates a unique 256-bit cryptographic fingerprint. No customer PII or bank numbers are exposed.

4
Blockchain Audit Registration

The hash is sent via Web3.py to the `TransactionAuditor` smart contract on Ethereum Sepolia, creating an immutable on-chain audit trail.

5
Tamper-Evident Verification

Auditors re-generate the SHA-256 hash from database records and query the smart contract. Any local alteration causes a cryptographic mismatch detection.
